cmake_minimum_required(VERSION 3.20)
project(reflect_pyrowave_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED PYROWAVE_SOURCE OR NOT DEFINED PYROWAVE_LIBRARY)
    message(FATAL_ERROR "Set PYROWAVE_SOURCE and PYROWAVE_LIBRARY to the pinned codec source and built C API library.")
endif()
add_executable(reflect-pyrowave-probe main.cpp)
target_include_directories(reflect-pyrowave-probe PRIVATE
    "${PYROWAVE_SOURCE}" "${PYROWAVE_SOURCE}/Granite/third_party/khronos/vulkan-headers/include")
target_link_libraries(reflect-pyrowave-probe PRIVATE "${PYROWAVE_LIBRARY}")
if(ANDROID)
    target_link_libraries(reflect-pyrowave-probe PRIVATE dl)
    target_link_options(reflect-pyrowave-probe PRIVATE "-Wl,-z,max-page-size=16384")
endif()
